\section*{1. Big-O Notation}


\[
f_1 = n^2 + n\log n + 3, \quad f_2 = 32(n+1)(n-1), \quad f_3 = n^2\log^2 n, \quad f_4 = \frac{5n^2}{\log n}(\log n + 2)^3, \quad f_5 = \sqrt{n}(n^{3/2} + 2)
\]

\begin{align*}
f_1 &= n^2 + n\log n + 3 \in \Theta(n^2) \\
f_2 &= 32(n^2 - 1) \in \Theta(n^2) \\
f_3 &= n^2 \log^2 n \in \Theta(n^2 \log^2 n) \\
f_4 &= 5n^2\log^2 n + 30n^2\log n + 60n^2 + \frac{40n^2}{\log n} \in \Theta(n^2 \log^2 n) \\
f_5 &= n^2 + 2\sqrt{n} \in \Theta(n^2)
\end{align*}

\[
\boxed{(f_1, f_2), \ (f_1, f_5), \ (f_2, f_5) \in \Theta(n^2) \qquad (f_3, f_4) \in \Theta(n^2 \log^2 n)}
\]


\section*{2. Ranking Asymptotic Growth Rates}


\[
n^4, \quad 2^{\sqrt{\log n}}, \quad n\log^2 n, \quad 2^n, \quad (n+1)!, \quad 2^n n!, \quad n2^n, \quad n\log n
\]

\begin{align*}
\log(2^{\sqrt{\log n}}) &= \sqrt{\log n} = o(\log n) \implies 2^{\sqrt{\log n}} = o(n^\epsilon) \\
\frac{n\log n}{n\log^2 n} &= \frac{1}{\log n} \to 0 \\
\frac{n^4}{2^n} &\to 0 \quad \text{(polynomial $<$ exponential)} \\
\frac{(n+1)!}{2^n \cdot n!} &= \frac{n+1}{2^n} \to 0
\end{align*}

\[
\boxed{2^{\sqrt{\log n}} < n\log n < n\log^2 n < n^4 < 2^n < n2^n < (n+1)! < 2^n n!}
\]

&\text{Conflicts:} \quad \{I_1, I_2\}, \{I_2, I_3\}, \{I_3, I_4\}, \{I_4, I_5\} \\
&\text{Graph:} \quad I_1 - I_2 - I_3 - I_4 - I_5 \quad (P_5)
\end{align*}
\begin{align*}
S^* &= \{I_1, I_3, I_5\}, \quad B_1 = 3 \le 4 = A_3, \quad B_3 = 7 \le 7 = A_5
\end{align*}
\[
\forall S \subseteq \{I_1, \ldots, I_5\},\ |S| = 4 \implies \exists \text{ conflict}
\]
\[
\boxed{S^* = \{(1,3), (4,7), (7,8)\}, \quad |S^*| = 3}
\]

\[
I_1 = (1,5), \quad I_2 = (0,2), \quad I_3 = (4,6)
