\section*{1. Online Hitting Set for Intervals}

\vspace{-10pt}
\subsection*{(a) Sophia: $\Omega(n)$}

$S = \{1, \ldots, n\}$. Adversary:
\[
A_i = [1,\, n{-}i{+}1], \quad i = 1, \ldots, n
\]

\textbf{Sophia:} $\forall\, i$: $A_i \cap H_{i-1} = \emptyset \implies$ add $\max(S \cap A_i) = n{-}i{+}1$.
\[
H_{\text{Sophia}} = \{1, 2, \ldots, n\}, \quad |H_{\text{Sophia}}| = n
\]

\textbf{OPT:} $1 \in A_i$ $\forall\, i \implies H^* = \{1\}$, $|H^*| = 1$.

\vspace{-8pt}
\[\boxed{\mathrm{ALG}/\mathrm{OPT} = n = \Omega(n)}\]

\vspace{-20pt}
\subsection*{(b) Tammy: $O(\log n)$}

Tammy adds $m = \lceil (\min(S \cap A) + \max(S \cap A))/2 \rceil$ when $A \cap H = \emptyset$.

Let $H^* = \{p_1^*, \ldots, p_k^*\}$ be OPT. $\forall\, A_i$ with $A_i \cap H_{i-1} = \emptyset$: $\exists\, p^* \in H^* \cap A_i$.

\textbf{Window:} $\forall\, p^* \in H^*$, define $W_{p^*} \subseteq S$ = points strictly between nearest added points flanking $p^*$. Initially $W_{p^*} = S$.

\textbf{Halving:} fix $p^* \in H^* \cap A_i$. $A_i \cap H_{i-1} = \emptyset \implies A_i \cap S \subseteq W_{p^*}$. $m$ midpoint of $A_i \cap S$, $p^* \in A_i \implies$
\[
|W_{p^*}^{\text{new}}| \le \tfrac{1}{2}|A_i \cap S| \le \tfrac{1}{2}|W_{p^*}^{\text{old}}|
\]

$|W_{p^*}| \le n$ initially $\implies \le \lceil \log_2 n \rceil$ such $A_i$ per $p^*$.
\[
|H_{\text{Tammy}}| \le \sum_{p^* \in H^*} \lceil \log_2 n \rceil = k \lceil \log_2 n \rceil
\]

\vspace{-8pt}
\[\boxed{\mathrm{ALG}/\mathrm{OPT} \le \lceil \log_2 n \rceil = O(\log n)}\]


\section*{2. Online Unit Covering}

\vspace{-10pt}
\subsection*{(i) Leftmost: unbounded}

\textbf{Adversary:} $\varepsilon \in (0, 1/n)$, $x_i = -(i-1)\varepsilon$.

$\forall\, j < i$: $x_j = -(j{-}1)\varepsilon > -(i{-}1)\varepsilon = x_i \implies x_i < x_j \implies x_i \notin [x_j, x_j{+}1]$. So Leftmost adds $[x_i, x_i{+}1]$ for every $i$.
\[
|\mathrm{ALG}| = n
\]

\textbf{OPT:} all $x_i \in [-(n{-}1)\varepsilon, 0] \subseteq [-(n{-}1)\varepsilon, 1{-}(n{-}1)\varepsilon]$ (length 1 since $(n-1)\varepsilon < 1$). $|\mathrm{OPT}| = 1$.

\vspace{-8pt}
\[\boxed{\mathrm{ALG}/\mathrm{OPT} = n \to \infty}\]

\vspace{-20pt}
\subsection*{(ii) Centered: 2-competitive}

Let $C = (c_1, \ldots, c_a)$ = centers in arrival order.

\textbf{Claim 1:} $\forall\, j < k$: $|c_k - c_j| > 1/2$.
\[
c_k \text{ uncovered} \wedge [c_j - \tfrac{1}{2}, c_j + \tfrac{1}{2}] \in \mathcal{C} \implies c_k \notin [c_j - \tfrac{1}{2}, c_j + \tfrac{1}{2}] \implies |c_k - c_j| > \tfrac{1}{2}
\]

\textbf{Claim 2:} $\forall\, I$ unit interval: $|C \cap I| \le 2$.
\[
c_p < c_q < c_r \in I \implies c_r - c_p > 1 \quad (\text{Claim 1: } c_q{-}c_p > \tfrac{1}{2},\, c_r{-}c_q > \tfrac{1}{2}) \quad \text{but } c_r - c_p \le 1 \;\bot
\]

$\forall\, c \in C: \exists\, I^* \in \mathrm{OPT}$ with $c \in I^*$ (OPT covers $c$). Each $I^*$ contains $\le 2$ centers.

\vspace{-8pt}
\[\boxed{\mathrm{ALG} = |C| \le 2 \cdot |\mathrm{OPT}|}\]

\vspace{-20pt}
\subsection*{(iii) Grid: 2-competitive}

Cell $k = [k, k{+}1)$, $k \in \mathbb{Z}$.

\textbf{Claim:} $\forall\, I^* = [a, a{+}1] \in \mathrm{OPT}$: $|\{k : I^* \cap \text{cell } k \ne \emptyset\}| \le 2$.
\[
|I^*| = 1 \implies I^* \subseteq \text{cell}\, \lfloor a \rfloor \cup \text{cell}\, (\lfloor a \rfloor + 1)
\]

ALG opens $\le 1$ interval per cell. $\forall\, x \in \mathrm{points}: \exists\, I^* \ni x \implies $ each ALG cell mapped to some OPT interval, each OPT interval $\to \le 2$ cells.

\vspace{-8pt}
\[\boxed{\mathrm{ALG} \le 2 \cdot \mathrm{OPT}}\]


\section*{3. Online Unit Clustering}

\vspace{-10pt}
\subsection*{Algorithm (Grid)}

\begin{algorithm}[H]
\DontPrintSemicolon
\SetKwBlock{Begin}{begin}{}
\Begin{
\textsc{GridCluster}$(x_1, x_2, \ldots)$\;
$\Pi \longleftarrow \emptyset$ \tcp*{partition into clusters, indexed by integer label}
\For{$i \longleftarrow 1, 2, \ldots$}{
    $k \longleftarrow \lfloor x_i \rfloor$\;
    \eIf{$\exists\, C_k \in \Pi$ with label $k$}{
        $C_k \longleftarrow C_k \cup \{x_i\}$\;
    }{
        $\Pi \longleftarrow \Pi \cup \{(\{x_i\}, k)\}$ \tcp*{new cluster, label $k$}
    }
}
\Return{$\Pi$}\;
}
\end{algorithm}

\textbf{Validity:} cluster labeled $k$ has all points $\in [k, k{+}1) \implies$ diameter $< 1 \implies$ coverable by $[k, k{+}1]$.

\vspace{-10pt}
\subsection*{Competitive analysis}

OPT: $k^*$ clusters $C_1^*, \ldots, C_{k^*}^*$. $\forall\, j$: $\exists\, a_j \in \mathbb{R}$ with $C_j^* \subseteq [a_j, a_j{+}1]$.

\textbf{Claim:} $\forall\, j$: $|\{\lfloor x \rfloor : x \in C_j^*\}| \le 2$.
\[
x \in C_j^* \subseteq [a_j, a_j{+}1] \implies \lfloor x \rfloor \in \{\lfloor a_j \rfloor,\, \lfloor a_j \rfloor + 1\}
\]

$|\mathrm{ALG}| = |\{\lfloor x_i \rfloor\}_i|$. Each OPT cluster $\to \le 2$ ALG labels:
\[
|\mathrm{ALG}| \le \sum_{j=1}^{k^*} 2 = 2 k^*
\]

\vspace{-8pt}
\[\boxed{\mathrm{ALG}/\mathrm{OPT} \le 2}\]
